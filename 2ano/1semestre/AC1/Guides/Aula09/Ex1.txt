
	.eqv read_int, 5

	#	$f2:	res;

	la	$t0, k1			# 
	l.s	$f4, 0($t0)		# $f4 = k1;
	la	$t0, k2			#
	l.s	$f6, 0($t0)		# $f6 = k2;
					#
do:					# do {
	li	$v0, read_int		#
	syscall				#	$v0 = read_int();
	move	$t0, $v0		#	$t0 = val;
					#

	cvt.s.w	$f0, $f0		#	$f0 = (float)val;

	syscall				#	print_float(res);
